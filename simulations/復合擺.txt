function getState(time) {
  const omega0 = getOmega0();
  const wPlus = omega0;
  const wMinus = omega0 * Math.sqrt(1 + 2 * coupling);
  const envelope = Math.exp(-damping * time);

  const qPlus = Aplus * envelope * Math.cos(wPlus * time);
  const qMinus = Aminus * envelope * Math.cos(wMinus * time);

  const theta1 = qPlus + qMinus;
  const theta2 = qPlus - qMinus;

  return { theta1, theta2, qPlus, qMinus, wPlus, wMinus };
}

function computeEnergy(time) {
  const omega0 = getOmega0();
  const wPlus = omega0;
  const wMinus = omega0 * Math.sqrt(1 + 2 * coupling);

  const cosP = Math.cos(wPlus * time);
  const sinP = Math.sin(wPlus * time);
  const cosM = Math.cos(wMinus * time);
  const sinM = Math.sin(wMinus * time);

  const decay = Math.exp(-2 * damping * time);

  const qPlusIdeal = Aplus * cosP;
  const qMinusIdeal = Aminus * cosM;

  const qPlusDotIdeal = -Aplus * wPlus * sinP;
  const qMinusDotIdeal = -Aminus * wMinus * sinM;

  const kinetic =
    decay *
    m1 * pendulumLength * pendulumLength *
    (
      qPlusDotIdeal * qPlusDotIdeal +
      qMinusDotIdeal * qMinusDotIdeal
    );

  const gravityPotential =
    decay *
    m1 * g * pendulumLength *
    (
      qPlusIdeal * qPlusIdeal +
      qMinusIdeal * qMinusIdeal
    );

  const springPotential =
    decay *
    2 * m1 * g * pendulumLength * coupling *
    qMinusIdeal * qMinusIdeal;

  const potential = gravityPotential + springPotential;
  const total = kinetic + potential;

  return {
    kinetic,
    gravityPotential,
    springPotential,
    potential,
    total
  };
}

function drawSpring(x1, y1, x2, y2) {
  const coils = 12;
  const amp = 6;
  const dx = x2 - x1;
  const dy = y2 - y1;
  const len = Math.sqrt(dx * dx + dy * dy);
  if (len < 1) return;

  const px = -dy / len;
  const py = dx / len;

  ctx.beginPath();

  for (let i = 0; i <= coils; i++) {
    const s = i / coils;
    const zig = i === 0 || i === coils ? 0 : (i % 2 === 0 ? amp : -amp);
    const x = x1 + dx * s + px * zig;
    const y = y1 + dy * s + py * zig;
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  }

  ctx.strokeStyle = "#e74c3c";
  ctx.lineWidth = 2.3;
  ctx.stroke();
}

function getDrawLengthPx() {
  const minM = 0.10;
  const maxM = 1.00;
  const minPx = 70;
  const maxPx = 355;

  const ratio = (pendulumLength - minM) / (maxM - minM);
  return minPx + ratio * (maxPx - minPx);
}

function getBobRadius1() {
  const minKg = 0.10;
  const maxKg = 1.00;
  const minR = 12;
  const maxR = 27;

  const ratio = (m1 - minKg) / (maxKg - minKg);
  return minR + ratio * (maxR - minR);
}

function getBobRadius2() {
  const minKg = 0.10;
  const maxKg = 1.00;
  const minR = 12;
  const maxR = 27;

  const ratio = (m2 - minKg) / (maxKg - minKg);
  return minR + ratio * (maxR - minR);
}

function drawMain(state) {
  const { theta1, theta2 } = state;

  ctx.clearRect(0, 0, canvas.width, canvas.height);

  const barY = 32;
  const barH = 13;
  const barBottom = barY + barH;

  const pivot1X = canvas.width * 0.34;
  const pivot2X = canvas.width * 0.61;

  const Lpx = getDrawLengthPx();
  const bobRadius1 = getBobRadius1();
  const bobRadius2 = getBobRadius2();

  const x1 = pivot1X + Lpx * Math.sin(theta1);
  const y1 = barBottom + Lpx * Math.cos(theta1);
  const x2 = pivot2X + Lpx * Math.sin(theta2);
  const y2 = barBottom + Lpx * Math.cos(theta2);

  ctx.fillStyle = "#34495e";
  ctx.fillRect(canvas.width * 0.23, barY, canvas.width * 0.55, barH);

  ctx.fillStyle = "#2c3e50";
  ctx.beginPath();
  ctx.arc(pivot1X, barBottom, 5, 0, Math.PI * 2);
  ctx.arc(pivot2X, barBottom, 5, 0, Math.PI * 2);
  ctx.fill();

  ctx.strokeStyle = "#455a64";
  ctx.lineWidth = 2.5;

  ctx.beginPath();
  ctx.moveTo(pivot1X, barBottom);
  ctx.lineTo(x1, y1);
  ctx.moveTo(pivot2X, barBottom);
  ctx.lineTo(x2, y2);
  ctx.stroke();

  drawSpring(x1, y1, x2, y2);

  ctx.fillStyle = "#e67e22";
  ctx.beginPath();
  ctx.arc(x1, y1, bobRadius1, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = "#2f80ed";
  ctx.beginPath();
  ctx.arc(x2, y2, bobRadius2, 0, Math.PI * 2);
  ctx.fill();

  ctx.fillStyle = "white";
  ctx.font = `bold ${Math.max(14, bobRadius1 * 0.75)}px Arial`;
  ctx.textAlign = "center";
  ctx.textBaseline = "middle";
  ctx.fillText("1", x1, y1);
  ctx.fillText("2", x2, y2);

  ctx.textBaseline = "alphabetic";
  ctx.fillStyle = "#1c2833";
  ctx.font = "bold 24px Arial";
  ctx.fillText("擺 1", pivot1X, barY - 8);
  ctx.fillText("擺 2", pivot2X, barY - 8);
}

function drawModalVector(state) {
  const W = modalVectorCanvas.width;
  const H = modalVectorCanvas.height;

  const xMaxDeg = 60;
  const yMaxDeg = 60;
  const padL = 80;
  const padR = 115;
  const padT = 40;
  const padB = 40;

  const plotW = W - padL - padR;
  const plotH = H - padT - padB;

  function xMap(deg) {
    return padL + ((deg + xMaxDeg) / (2 * xMaxDeg)) * plotW;
  }

  function yMap(deg) {
    return padT + ((yMaxDeg - deg) / (2 * yMaxDeg)) * plotH;
  }

  const xMin = xMap(-xMaxDeg);
  const xMax = xMap(xMaxDeg);
  const yTop = yMap(yMaxDeg);
  const yBottom = yMap(-yMaxDeg);
  const cx = xMap(0);
  const cy = yMap(0);

  mvctx.clearRect(0, 0, W, H);
  mvctx.fillStyle = "#fbfcfd";
  mvctx.fillRect(0, 0, W, H);

  mvctx.strokeStyle = "#e2e8ee";
  mvctx.lineWidth = 1;

  for (let d = -60; d <= 60; d += 20) {
    const x = xMap(d);
    const y = yMap(d);

    mvctx.beginPath();
    mvctx.moveTo(x, yTop);
    mvctx.lineTo(x, yBottom);
    mvctx.stroke();

    mvctx.beginPath();
    mvctx.moveTo(xMin, y);
    mvctx.lineTo(xMax, y);
    mvctx.stroke();
  }

  mvctx.strokeStyle = "#e7edf3";
  mvctx.lineWidth = 2;
  mvctx.strokeRect(xMin, yTop, plotW, plotH);

  mvctx.strokeStyle = "#455a64";
  mvctx.lineWidth = 2.2;
  mvctx.beginPath();
  mvctx.moveTo(xMin, cy);
  mvctx.lineTo(xMax, cy);
  mvctx.moveTo(cx, yBottom);
  mvctx.lineTo(cx, yTop);
  mvctx.stroke();

  mvctx.fillStyle = "#1c2833";
  mvctx.font = "bold 20px Arial";
  mvctx.textBaseline = "middle";

  mvctx.textAlign = "left";
  mvctx.fillText("q+ (°)", xMax + 20, cy);

  mvctx.textAlign = "center";
  mvctx.fillText("q− (°)", cx, yTop - 26);

  mvctx.fillStyle = "#37474f";
  mvctx.font = "bold 18px Arial";

  mvctx.textAlign = "center";
  mvctx.fillText("-60°", xMin, cy + 36);
  mvctx.fillText("+60°", xMax, cy + 36);

  mvctx.textAlign = "right";
  mvctx.fillText("+60°", cx - 12, yTop - 2);
  mvctx.fillText("-60°", cx - 12, yBottom + 18);

  const qx = radToDeg(state.qPlus);
  const qy = radToDeg(state.qMinus);

  const x = xMap(qx);
  const y = yMap(qy);

  mvctx.strokeStyle = "#1f3a5f";
  mvctx.lineWidth = 3.2;
  mvctx.beginPath();
  mvctx.moveTo(cx, cy);
  mvctx.lineTo(x, y);
  mvctx.stroke();

  mvctx.setLineDash([6, 5]);
  mvctx.strokeStyle = "#9aa7b2";
  mvctx.beginPath();
  mvctx.moveTo(x, cy);
  mvctx.lineTo(x, y);
  mvctx.moveTo(cx, y);
  mvctx.lineTo(x, y);
  mvctx.stroke();
  mvctx.setLineDash([]);

  mvctx.fillStyle = "#e67e22";
  mvctx.beginPath();
  mvctx.arc(x, cy, 8, 0, Math.PI * 2);
  mvctx.fill();

  mvctx.fillStyle = "#8e44ad";
  mvctx.beginPath();
  mvctx.arc(cx, y, 8, 0, Math.PI * 2);
  mvctx.fill();

  mvctx.fillStyle = "#1f3a5f";
  mvctx.beginPath();
  mvctx.arc(x, y, 10, 0, Math.PI * 2);
  mvctx.fill();

  mvctx.font = "bold 16px Arial";
  mvctx.textAlign = "left";
  mvctx.fillStyle = "#1f3a5f";

function drawHistoryGraph() {
  const W = historyCanvas.width;
  const H = historyCanvas.height;

  const padL = 92;
  const padR = 28;
  const padT = 30;
  const padB = 48;

  const plotW = W - padL - padR;
  const plotH = H - padT - padB;
  const midY = padT + plotH / 2;
  const maxDeg = 90;

  hctx.clearRect(0, 0, W, H);
  hctx.fillStyle = "#fbfcfd";
  hctx.fillRect(0, 0, W, H);

  hctx.strokeStyle = "#dbe3ea";
  hctx.lineWidth = 1;
  hctx.fillStyle = "#37474f";
  hctx.font = "bold 22px Arial";
  hctx.textAlign = "right";

  for (let d = -90; d <= 90; d += 30) {
    const y = midY - (d / maxDeg) * (plotH / 2);
    hctx.beginPath();
    hctx.moveTo(padL, y);
    hctx.lineTo(W - padR, y);
    hctx.stroke();
    hctx.fillText(`${d}°`, padL - 10, y + 7);
  }

  hctx.strokeStyle = "#607d8b";
  hctx.lineWidth = 1.8;
  hctx.beginPath();
  hctx.moveTo(padL, midY);
  hctx.lineTo(W - padR, midY);
  hctx.moveTo(padL, padT);
  hctx.lineTo(padL, H - padB);
  hctx.stroke();

  hctx.fillStyle = "#1c2833";
  hctx.font = "bold 24px Arial";
  hctx.textAlign = "center";
  hctx.fillText("時間 t (s)", padL + plotW / 2, H - 14);

  hctx.save();
  hctx.translate(30, padT + plotH / 2);
  hctx.rotate(-Math.PI / 2);
  hctx.fillText("角度 (度)", 0, 0);
  hctx.restore();

  function yMap(rad) {
    return midY - (radToDeg(rad) / maxDeg) * (plotH / 2);
  }

  function drawLine(key, color) {
    if (history.length < 2) return;

    hctx.beginPath();

    history.forEach((p, i) => {
      const x = padL + (i / (historyMax - 1)) * plotW;
      const y = yMap(p[key]);
      if (i === 0) hctx.moveTo(x, y);
      else hctx.lineTo(x, y);
    });

    hctx.strokeStyle = color;
    hctx.lineWidth = 3.2;
    hctx.stroke();
  }

  drawLine("theta1", "#e67e22");
  drawLine("theta2", "#2f80ed");
  drawLine("qPlus", "#27ae60");
  drawLine("qMinus", "#8e44ad");
}

function drawEnergyGraph() {
  const W = energyCanvas.width;
  const H = energyCanvas.height;

  const padL = 98;
  const padR = 32;
  const padT = 36;
  const padB = 58;

  const plotW = W - padL - padR;
  const plotH = H - padT - padB;

  ectx.clearRect(0, 0, W, H);
  ectx.fillStyle = "#fbfcfd";
  ectx.fillRect(0, 0, W, H);

  if (history.length < 2) {
    ectx.fillStyle = "#263238";
    ectx.font = "bold 20px Arial";
    ectx.textAlign = "left";
    ectx.fillText("等待能量資料...", padL + 20, padT + 42);
    return;
  }

  const visible = history.slice(-historyMax);

  let maxE = 0;
  visible.forEach(p => {
    maxE = Math.max(
      maxE,
      p.kinetic,
      p.potential,
      p.gravityPotential,
      p.springPotential,
      p.total
    );
  });

  if (maxE <= 0) maxE = 1;
  maxE *= 1.08;

  const t0 = visible[0].t;
  const t1 = visible[visible.length - 1].t;
  const dt = Math.max(0.001, t1 - t0);

  ectx.fillStyle = "#37474f";
  ectx.font = "bold 18px Arial";
  for (let i = 0; i <= 4; i++) {
    const y = padT + plotH * i / 4;
    const val = maxE * (1 - i / 4);

    ectx.strokeStyle = "#dbe4ec";
    ectx.lineWidth = 1.2;
    ectx.beginPath();
    ectx.moveTo(padL, y);
    ectx.lineTo(padL + plotW, y);
    ectx.stroke();

    ectx.textAlign = "right";
    ectx.fillText(val.toFixed(2), padL - 10, y + 6);
  }

  ectx.strokeStyle = "#78909c";
  ectx.lineWidth = 1.8;
  ectx.beginPath();
  ectx.moveTo(padL, padT);
  ectx.lineTo(padL, padT + plotH);
  ectx.lineTo(padL + plotW, padT + plotH);
  ectx.stroke();

  function mapX(p) {
    return padL + ((p.t - t0) / dt) * plotW;
  }

  function mapY(val) {
    return padT + plotH - (val / maxE) * plotH;
  }

  function plot(key, color, width, dashed = false) {
    ectx.save();
    ectx.setLineDash(dashed ? [9, 7] : []);
    ectx.beginPath();

    visible.forEach((p, i) => {
      const x = mapX(p);
      const y = mapY(p[key]);
      if (i === 0) ectx.moveTo(x, y);
      else ectx.lineTo(x, y);
    });

    ectx.strokeStyle = color;
    ectx.lineWidth = width;
    ectx.stroke();
    ectx.restore();
  }

  plot("kinetic", "#ef5350", 3.2, false);
  plot("gravityPotential", "#66bb6a", 3.0, true);
  plot("springPotential", "#ab47bc", 3.0, true);
  plot("potential", "#42a5f5", 3.2, false);
  plot("total", "#263238", 3.6, false);

  ectx.fillStyle = "#1c2833";
  ectx.font = "bold 24px Arial";
  ectx.textAlign = "center";
  ectx.fillText("時間 t (s)", padL + plotW / 2, H - 22);

  ectx.save();
  ectx.translate(28, padT + plotH / 2);
  ectx.rotate(-Math.PI / 2);
  ectx.fillText("能量 (J)", 0, 0);
  ectx.restore();
}

function updateTable(state) {
  theta1Cell.textContent = radToDeg(state.theta1).toFixed(1) + "°";
  theta2Cell.textContent = radToDeg(state.theta2).toFixed(1) + "°";
  qPlusCell.textContent = radToDeg(state.qPlus).toFixed(1) + "°";
  qMinusCell.textContent = radToDeg(state.qMinus).toFixed(1) + "°";
}

function updateMainOutsideInfo(state) {
  qPlusOutside.innerHTML = `q<sub>+</sub> = ${radToDeg(state.qPlus).toFixed(1)}°`;
  qMinusOutside.innerHTML = `q<sub>-</sub> = ${radToDeg(state.qMinus).toFixed(1)}°`;
  timeOutside.textContent = `時間 t = ${t.toFixed(1)} s`;
  lengthOutside.textContent = `繩長 L = ${Math.round(pendulumLength * 100)} cm`;
  massOutside.textContent = `m₁ = ${Math.round(m1 * 1000)} g, m₂ = ${Math.round(m2 * 1000)} g`;
  dampingOutside.textContent = `阻尼 b = ${damping.toFixed(2)}`;
  omegaOutside.innerHTML =
    `ω<sub>+</sub> = ${state.wPlus.toFixed(2)}, ω<sub>-</sub> = ${state.wMinus.toFixed(2)} rad/s`;
}

function loop() {
  const state = getState(t);
  const energy = computeEnergy(t);

  drawMain(state);
  drawModalVector(state);
  drawHistoryGraph();
  drawEnergyGraph();
  updateTable(state);
  updateMainOutsideInfo(state);

  if (running) {
    history.push({
      t,
      theta1: state.theta1,
      theta2: state.theta2,
      qPlus: state.qPlus,
      qMinus: state.qMinus,
      kinetic: energy.kinetic,
      gravityPotential: energy.gravityPotential,
      springPotential: energy.springPotential,
      potential: energy.potential,
      total: energy.total
    });

    if (history.length > historyMax) history.shift();

    t += 0.006 * speed;
  }

  requestAnimationFrame(loop);
}

setValuesFromSliders();
